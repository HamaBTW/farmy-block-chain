# Farmy v2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Turn Farmy into v2: land, machine and lot NFTs, lots that split by quantity, check-and-validate in one step, and an ETH-backed, capped FarmCoin. The web app and docs follow.

**Architecture:** `BatchRegistry` stays one ERC-721, but every token gets a `Kind` (Land, Machine, Lot). Lots come from land through `harvest` and split into child lots through `splitLot`, which closes the parent. `IBatchRegistry` v2 adds `isActive`, so the escrow refuses closed lots. `FarmCoin` drops the owner's `mint` and gets `buy`/`sell` at 1 ETH = 1000 FRM, with a cap set at deploy time. The Next.js app reads the new views and events.

**Tech Stack:** Solidity 0.8.24 (evm `paris`), OpenZeppelin 5.0.2, Truffle 5.11 + Ganache 7 (`127.0.0.1:7545`, network 5777, chain 1337), Next.js 16 + React + ethers v6 + Tailwind, `node --test` for UI unit tests.

**Spec:** `docs/superpowers/specs/2026-11-08-farmy-v2-assets-lots-ethcoin-design.md`

- Read `rules.md` before coding. NatSpec on every external function, custom errors, cheapest check first, checks → effects → interactions.
- solc pinned `0.8.24`, `evmVersion: "paris"`, optimizer 200 (already in `farmy_blockchain/truffle-config.js`); OpenZeppelin stays `5.0.2`. **No new npm dependencies.**

- Truffle can't decode custom errors: assert reverts with `expectRevert(promise, "ErrorName(types)")` from the test helpers.

- UI code: write `BigInt(0)`, never `0n` (tsconfig targets ES2017). Files imported by `*.test.ts` import siblings with the `.ts` extension.
- Plain words in UI copy (see `farmy_web/DESIGN_SYSTEM.md`). The app calls the registry **Farm**; the contract keeps the name `BatchRegistry`.

- FarmCoin: `RATE = 1000` (1 wei → 1000 base units, so 1 ETH = 1000 FRM); the migration deploys with a cap of **1,000,000 FRM**; the escrow's max markup stays 20%.

```solidity
// SPDX-License-Identifier: MIT
pragma solidity 0.8.24;

import {ERC20} from "@openzeppelin/contracts/token/ERC20/ERC20.sol";
import {ERC20Capped} from "@openzeppelin/contracts/token/ERC20/extensions/ERC20Capped.sol";
import {ReentrancyGuard} from "@openzeppelin/contracts/utils/ReentrancyGuard.sol";

/// @title FarmCoin
/// @notice Payment token (FRM) for Farmy, backed 1:1000 by ETH held in this contract.
///         Anyone buys FRM with ETH and sells it back at the same fixed rate. Test chain only.
/// @dev No owner and no free mint: FRM only comes from `buy`. Invariant: reserve == totalSupply / RATE.
contract FarmCoin is ERC20Capped, ReentrancyGuard {
    /// @notice FRM base units per wei: 1 ETH = 1000 FRM.
    uint256 public constant RATE = 1000;

    event Bought(address indexed buyer, uint256 ethIn, uint256 frmOut);
    event Sold(address indexed seller, uint256 frmIn, uint256 ethOut);

    error ZeroAmount();
    error EthTransferFailed();

    /// @param maxSupply Max FRM supply in base units (the migration uses 1,000,000 FRM).
    constructor(uint256 maxSupply) ERC20("FarmCoin", "FRM") ERC20Capped(maxSupply) {}

    /// @notice Buys FRM with the sent ETH at the fixed rate.
    /// @return frmOut FRM minted to the caller, in base units.
    function buy() external payable returns (uint256 frmOut) {
        if (msg.value == 0) revert ZeroAmount();
        frmOut = msg.value * RATE;
        _mint(msg.sender, frmOut); // ERC20Capped reverts ERC20ExceededCap above the cap
        emit Bought(msg.sender, msg.value, frmOut);
    }

    /// @notice Sells FRM back for ETH from the reserve. Only whole wei are paid; the rest stays.
    /// @param amount FRM in base units the caller wants to sell.
    /// @return ethOut Wei sent to the caller.
    function sell(uint256 amount) external nonReentrant returns (uint256 ethOut) {
        ethOut = amount / RATE;
        if (ethOut == 0) revert ZeroAmount();
        uint256 frmIn = ethOut * RATE;
        _burn(msg.sender, frmIn);
        emit Sold(msg.sender, frmIn, ethOut);
        (bool ok, ) = msg.sender.call{value: ethOut}("");
        if (!ok) revert EthTransferFailed();
    }

    /// @notice ETH held to back every FRM.
    function reserve() external view returns (uint256) {
        return address(this).balance;
    }
}
```

```solidity
/// @notice One NFT per farm asset: land parcels, big machines, and lots of produce harvested from land.
///         Tracks custody, inspection, validation and the last declared price. Holds no money. Test chain only.
/// @dev The NFT owner is the custodian. The contract owner is the regulator (also the land office).
contract BatchRegistry is ERC721URIStorage, Ownable, IBatchRegistry {
    enum Kind {
        Land,
        Machine,
        Lot
    }

    /// @dev Slot 1: lastPrice, validated, kind, closed, landId (27 bytes). Slot 2: parentId, quantity (24 bytes).
    struct Asset {
        uint128 lastPrice;
        bool validated;
        Kind kind;
        bool closed;
        uint64 landId;
        uint64 parentId;
        uint128 quantity;

    mapping(address => bool) private _inspectors;

    uint256 private _lastId;

    event LotHarvested(uint256 indexed batchId, uint256 indexed landId, uint256 quantity);
    event CustodyTransferred(uint256 indexed batchId, address indexed from, address indexed to);
    event CheckpointAdded(uint256 indexed batchId, address indexed inspector, bytes32 dataHash);
    event BatchValidated(uint256 indexed batchId, address indexed inspector);
    event PriceRecorded(uint256 indexed batchId, uint256 price);

    error ZeroAddress();
    error ZeroPrice();
    error NotInspector();
    error AlreadyValidated();
    error NotEscrow();
    error WrongKind();
    error NotCustodian();
    error ZeroQuantity();
    error LotClosed();

    modifier onlyInspector() {
        if (!_inspectors[msg.sender]) revert NotInspector();
        _;
    }

    /// @notice Deploys the registry with the deployer as regulator.
    constructor() ERC721("Farmy Batch", "FBATCH") Ownable(msg.sender) {}

    /// @notice Registers a land parcel or a machine (the regulator acts as the land office).
    /// @param to First holder (usually the farmer).
    /// @param kind Land or Machine (lots come from `harvest`).
    /// @param uri Off-chain details URI or hash.
    /// @param initialPrice Declared price in FarmCoin base units.
    /// @return id The new token id (ids start at 1 and are shared by all kinds).
    function registerAsset(address to, Kind kind, string calldata uri, uint256 initialPrice) external onlyOwner returns (uint256 id) {
        if (kind == Kind.Lot) revert WrongKind();
        id = _create(to, kind, uri, initialPrice);
    }

    /// @param landId The land token the produce comes from.
    /// @param uri Off-chain details (product, unit, harvest date) URI or hash.
    /// @param quantity Whole units (the unit lives in the details), must be > 0.
    /// @param initialPrice Declared price of the whole lot in FarmCoin base units.
    /// @return lotId The new lot id.
    function harvest(uint256 landId, string calldata uri, uint256 quantity, uint256 initialPrice) external returns (uint256 lotId) {
        if (_requireOwned(landId) != msg.sender) revert NotCustodian();
        if (_assets[landId].kind != Kind.Land) revert WrongKind();
        if (quantity == 0) revert ZeroQuantity();
        lotId = _create(msg.sender, Kind.Lot, uri, initialPrice);
        Asset storage lot = _assets[lotId];
        lot.landId = SafeCast.toUint64(landId);
        lot.quantity = SafeCast.toUint128(quantity);
        emit LotHarvested(lotId, landId, quantity);
    }

    /// @notice Hands the asset (and the NFT) to the next holder. Closed lots can't move.
    /// @param batchId The token id.
    /// @param newCustodian Next holder.
    function transferCustody(uint256 batchId, address newCustodian) external {
        // OZ checks: token exists, caller is the holder, receiver isn't zero. `_update` refuses closed lots.
        _transfer(msg.sender, newCustodian, batchId);

    /// @param batchId The token id.
    /// @param dataHash Hash of the off-chain record (weight, temperature, place, time).
    function addCheckpoint(uint256 batchId, bytes32 dataHash) external onlyInspector {
        _active(batchId);
        emit CheckpointAdded(batchId, msg.sender, dataHash);

    /// @param batchId The token id.

        emit CheckpointAdded(batchId, msg.sender, dataHash);
        emit BatchValidated(batchId, msg.sender);

    /// @param batchId The token id.

        if (price == 0) revert ZeroPrice();
        _requireOwned(batchId);
        _assets[batchId].lastPrice = SafeCast.toUint128(price);

    /// @param account Inspector address.
    /// @param allowed True to grant, false to revoke.
    function setInspector(address account, bool allowed) external onlyOwner {
        if (account == address(0)) revert ZeroAddress();
        _inspectors[account] = allowed;
    }

    /// @notice Sets the only address allowed to call `recordSalePrice` (the PriceCapEscrow).
    /// @param escrowAddress Deployed escrow contract.
    function setEscrow(address escrowAddress) external onlyOwner {
        if (escrowAddress == address(0)) revert ZeroAddress();
        _escrow = escrowAddress;
    }

    /// @notice Whether `account` is an inspector.
    function isInspector(address account) external view returns (bool) {
        return _inspectors[account];
    }

    /// @notice Kind, links and quantity of a token.
    /// @param id The token id.
    /// @return kind Land, Machine or Lot.
    /// @return landId Land a lot comes from (0 for land and machines).
    /// @return parentId Lot this lot was split from (0 if none).
    /// @return quantity Units in a lot (0 for land and machines).
    /// @return closed True once a lot was split.
    function assetInfo(uint256 id)
        external
        view
        returns (Kind kind, uint256 landId, uint256 parentId, uint256 quantity, bool closed)
    {
        _requireOwned(id);
        Asset storage a = _assets[id];
        return (a.kind, a.landId, a.parentId, a.quantity, a.closed);
    }

    /// @inheritdoc IBatchRegistry
    function isActive(uint256 batchId) external view returns (bool) {
        _requireOwned(batchId);
        return !_assets[batchId].closed;
    }

    /// @inheritdoc IBatchRegistry
    function getLastPrice(uint256 batchId) external view returns (uint256) {
        _requireOwned(batchId);
        return _assets[batchId].lastPrice;
    }

    /// @inheritdoc IBatchRegistry
    function isValidated(uint256 batchId) external view returns (bool) {
        _requireOwned(batchId);
        return _assets[batchId].validated;
    }

    /// @inheritdoc IBatchRegistry
    function custodianOf(uint256 batchId) external view returns (address) {
        return _requireOwned(batchId);
    }

    /// @notice Address allowed to record sale prices.
    function escrow() external view returns (address) {
        return _escrow;
    }

    /// @dev Mints a token with its first price. Every new token emits AssetRegistered + PriceRecorded.
    function _create(address to, Kind kind, string memory uri, uint256 price) internal returns (uint256 id) {
        if (price == 0) revert ZeroPrice();
        unchecked {
            id = ++_lastId;
        }
        Asset storage a = _assets[id];
        a.lastPrice = SafeCast.toUint128(price);
        a.kind = kind;
        _mint(to, id);
        _setTokenURI(id, uri);
        emit AssetRegistered(id, kind, to);
        emit PriceRecorded(id, price);
    }

    /// @dev Existing, not-closed token.
    function _active(uint256 id) internal view returns (Asset storage a) {
        _requireOwned(id);
        a = _assets[id];
        if (a.closed) revert LotClosed();
    }

    /// @dev Every transfer is a custody change (plain ERC-721 transfers too). Closed lots never move.
    function _update(address to, uint256 tokenId, address auth) internal override returns (address from) {
        if (_assets[tokenId].closed) revert LotClosed();
        from = super._update(to, tokenId, auth);
        if (from != address(0)) emit CustodyTransferred(tokenId, from, to);
    }
```

```solidity
    /// @notice Splits a lot into 2–10 smaller lots so parts can go to different places.
    ///         The parent is closed (never burned, so its history stays). Each child keeps the land,
    ///         the parent link, the validation and the URI; the price is shared in proportion.
    /// @param lotId The lot to split; the caller must hold it.
    /// @param quantities Units of each child, each > 0, adding up to the lot's quantity.
    /// @return childIds The new lot ids, in the order of `quantities`.
    function splitLot(uint256 lotId, uint256[] calldata quantities) external returns (uint256[] memory childIds) {
        if (_requireOwned(lotId) != msg.sender) revert NotCustodian();
        Asset storage parent = _assets[lotId];
        if (parent.kind != Kind.Lot) revert WrongKind();
        if (parent.closed) revert LotClosed();
        uint256 parts = quantities.length;
        if (parts < 2 || parts > MAX_PARTS) revert BadSplit();
        uint256 total;
        for (uint256 i; i < parts; ++i) {
            if (quantities[i] == 0) revert BadSplit();
            total += quantities[i];
        }
        if (total != parent.quantity) revert BadSplit();

        parent.closed = true;
        uint256 parentPrice = parent.lastPrice;
        uint256 priceLeft = parentPrice;
        uint64 landId = parent.landId;
        bool validated = parent.validated;
        string memory uri = tokenURI(lotId);
        childIds = new uint256[](parts);
        for (uint256 i; i < parts; ++i) {
            uint256 price = i == parts - 1 ? priceLeft : (parentPrice * quantities[i]) / total;
            priceLeft -= price;
            uint256 childId = _create(msg.sender, Kind.Lot, uri, price);
            Asset storage child = _assets[childId];
            child.landId = landId;
            child.parentId = SafeCast.toUint64(lotId);
            child.quantity = SafeCast.toUint128(quantities[i]);
            child.validated = validated;
            childIds[i] = childId;
            emit LotSplit(lotId, childId, quantities[i]);
        }
    }
```

```ts
/** Same limit as BatchRegistry.MAX_PARTS. */
export const MAX_PARTS = 10;

/** Mirrors BatchRegistry.splitLot: price × part / total, the last part takes the rounding remainder. */
export function splitPrices(price: bigint, parts: bigint[]): bigint[] {
  const total = parts.reduce((a, q) => a + q, BigInt(0));
  let left = price;
  return parts.map((q, i) => {
    const p = i === parts.length - 1 ? left : (price * q) / total;
    left -= p;
    return p;
  });
}

/** The reason a split would be refused (BadSplit), in plain words, or null when it is fine. */
export function checkSplit(quantity: bigint, parts: bigint[]): string | null {
  if (parts.length < 2) return "Split into at least 2 parts.";
  if (parts.length > MAX_PARTS) return `Split into at most ${MAX_PARTS} parts.`;
  if (parts.some((q) => q <= BigInt(0))) return "Every part needs a quantity above 0.";
  const sum = parts.reduce((a, q) => a + q, BigInt(0));
  return sum === quantity ? null : `The parts add up to ${sum}, but the lot has ${quantity}.`;
}
```

```ts
import type { Contract, EventLog } from "ethers";
import type { AssetKind, JourneyState, OpenSale, Step } from "./journey";

/** Same order as BatchRegistry.Kind. */
export const KINDS: AssetKind[] = ["land", "machine", "lot"];

/** An asset as screens show it: its journey state, metadata URI and lot links. */
export type BatchView = JourneyState & { uri: string; landId: bigint; parentId: bigint; quantity: bigint };

type AssetInfo = { kind: AssetKind; landId: bigint; parentId: bigint; quantity: bigint; closed: boolean };

const OPEN_STATES: Record<number, OpenSale["state"]> = { 1: "Created", 2: "Deposited" };

async function loadAssetInfo(registry: Contract, id: bigint): Promise<AssetInfo> {
  const i = await registry.assetInfo(id);
  return { kind: KINDS[Number(i.kind)], landId: i.landId, parentId: i.parentId, quantity: i.quantity, closed: i.closed };
}

/** Live state of one asset and its open sale (escrow may be null if not deployed yet). */
export async function loadJourneyState(registry: Contract, escrow: Contract | null, batchId: bigint): Promise<JourneyState> {
  const [custodian, validated, lastPrice, saleId, info] = await Promise.all([
    registry.custodianOf(batchId) as Promise<string>,
    registry.isValidated(batchId) as Promise<boolean>,
    registry.getLastPrice(batchId) as Promise<bigint>,
    escrow ? (escrow.openSaleOf(batchId) as Promise<bigint>) : Promise.resolve(BigInt(0)),
    loadAssetInfo(registry, batchId),
  ]);
  let sale: OpenSale | null = null;
  if (escrow && saleId !== BigInt(0)) {
    const s = await escrow.getSale(saleId);
    const state = OPEN_STATES[Number(s.state)];
    if (state) sale = { id: saleId, seller: s.seller, buyer: s.buyer, price: s.price, state };
  }
  return { batchId, kind: info.kind, closed: info.closed, custodian, validated, lastPrice, sale };
}

/** Ids of every asset, newest first (the registry isn't enumerable; every new token emits AssetRegistered). */
export async function loadBatchIds(registry: Contract): Promise<bigint[]> {
  const logs = await registry.queryFilter(registry.filters.AssetRegistered());
  return logs.map((log) => (log as EventLog).args.batchId as bigint).reverse();
}

export async function loadBatchView(registry: Contract, escrow: Contract | null, batchId: bigint): Promise<BatchView> {
  const [state, uri, info] = await Promise.all([
    loadJourneyState(registry, escrow, batchId),
    registry.tokenURI(batchId) as Promise<string>,
    loadAssetInfo(registry, batchId),
  ]);
  return { ...state, uri, landId: info.landId, parentId: info.parentId, quantity: info.quantity };
}

export async function loadAllBatchViews(registry: Contract, escrow: Contract | null): Promise<BatchView[]> {
  const ids = await loadBatchIds(registry);
  return Promise.all(ids.map((id) => loadBatchView(registry, escrow, id)));
}

/** Lots a lot was split into, in split order. */
export async function loadChildren(registry: Contract, lotId: bigint): Promise<bigint[]> {
  const logs = await registry.queryFilter(registry.filters.LotSplit(lotId));
  return logs.map((log) => (log as EventLog).args.childId as bigint);
}

/** Lots harvested from a land parcel, oldest first. */
export async function loadHarvests(registry: Contract, landId: bigint): Promise<bigint[]> {
  const logs = await registry.queryFilter(registry.filters.LotHarvested(null, landId));
  return logs.map((log) => (log as EventLog).args.batchId as bigint);
}

/** What the journey page hands to the market panel (B) so it can render the sale steps of this asset. */
export type SaleContext = { batch: BatchView; steps: Step[]; me: string; onDone: () => void };
```

```tsx
import type { BatchView } from "@/lib/journeyData";
import { hashFromUri, type AssetDoc } from "@/lib/metadata";
import { Chip } from "./Chip";

/** Validation, sale and split chips for an asset, in the status colors of DESIGN_SYSTEM.md §2. */
export function BatchChips({ batch }: { batch: BatchView }) {
  if (batch.closed) return <Chip tone="plain">✂️ Split</Chip>;
  return (
    <>
      {batch.validated ? <Chip tone="field">✅ Validated</Chip> : <Chip tone="plain">Not validated</Chip>}
      {batch.sale?.state === "Created" && <Chip tone="harvest">🏷️ For sale</Chip>}
      {batch.sale?.state === "Deposited" && <Chip tone="harvest">🔒 Paid into escrow</Chip>}
    </>
  );
}

/** The off-chain description of an asset (land, machine or lot), when this browser has it. */
export function assetDocOf(uri: string): AssetDoc | null {
  const hash = hashFromUri(uri);
  const doc = hash ? getDoc(hash) : null;
  return doc && doc.schema !== "farmy.checkpoint.v1" ? doc : null;
}
```

```ts
import type { Contract, EventLog, Log } from "ethers";

export type HistoryKind = "AssetRegistered" | "LotHarvested" | "LotSplit" | "CustodyTransferred" | "CheckpointAdded" | "BatchValidated" | "PriceRecorded";

export type HistoryItem = {
  kind: HistoryKind;
  blockNumber: number;
  logIndex: number;
  txHash: string;
  timestamp: number;
  /** Every event argument as a string, including `batchId`. */
  args: Record<string, string>;
};

/** Every registry event about one asset, oldest first: its own events plus harvests from it (land) and splits into/out of it (lots). */
export async function loadHistory(registry: Contract, id: bigint): Promise<HistoryItem[]> {
  const f = registry.filters;
  const filters = [
    f.AssetRegistered(id),
    f.LotHarvested(id),
    f.LotHarvested(null, id),
    f.LotSplit(id),
    f.LotSplit(null, id),
    f.CustodyTransferred(id),
    f.CheckpointAdded(id),
    f.BatchValidated(id),
    f.PriceRecorded(id),
  ];
  const perFilter = await Promise.all(filters.map((filter) => registry.queryFilter(filter)));
  const seen = new Set<string>();
  const logs = (perFilter.flat() as EventLog[]).filter((l) => {
    const key = `${l.transactionHash}-${l.index}`;
    if (seen.has(key)) return false;
    seen.add(key);
    return true;
  });
  logs.sort((a, b) => a.blockNumber - b.blockNumber || a.index - b.index);

  const timestamps = new Map<number, number>();
  await Promise.all(
    [...new Set(logs.map((l) => l.blockNumber))].map(async (n) => {
      const block = await registry.runner!.provider!.getBlock(n);
      timestamps.set(n, block?.timestamp ?? 0);
    }),
  );

  return logs.map((log) => ({
    kind: log.eventName as HistoryKind,
    blockNumber: log.blockNumber,
    logIndex: log.index,
    txHash: log.transactionHash,
    timestamp: timestamps.get(log.blockNumber) ?? 0,
    args: argsOf(log),
  }));
}

function argsOf(log: EventLog | Log): Record<string, string> {
  if (!("fragment" in log)) return {};
  const out: Record<string, string> = {};
  log.fragment.inputs.forEach((input, i) => {
    out[input.name] = String(log.args[i]);
  });
  return out;
}
```

```tsx
/** Registry events as stamps. A token's first PriceRecorded is folded into its creation stamp. */
function registryStamps(history: HistoryItem[], id: bigint, unit: string): StampItem[] {
  const self = id.toString();
  const createdTx = new Set(history.filter((h) => h.kind === "AssetRegistered" && h.args.batchId === self).map((h) => h.txHash));
  const coveredTx = new Set(
    history.filter((h) => (h.kind === "LotHarvested" && h.args.batchId === self) || (h.kind === "LotSplit" && h.args.childId === self)).map((h) => h.txHash),
  );
  const firstPrice = new Map(history.filter((h) => h.kind === "PriceRecorded" && createdTx.has(h.txHash)).map((h) => [h.txHash, h.args.price]));
  const withPrice = (tx: string, text: string) => (firstPrice.has(tx) ? `${text} · first price ${formatFrm(firstPrice.get(tx)!)}` : text);
  const lotLink = (lot: string) => (
    <Link href={`/batch/${lot}`} className="underline">
      lot #{lot}
    </Link>
  );

  return history.flatMap((h): StampItem[] => {
    const base = { key: `r-${h.blockNumber}-${h.logIndex}`, blockNumber: h.blockNumber, logIndex: h.logIndex, timestamp: h.timestamp };
    switch (h.kind) {
      case "AssetRegistered":
        return coveredTx.has(h.txHash)
          ? []
          : [{ ...base, emoji: "🌱", tone: "field", title: <>Registered to <PersonName address={h.args.to} showAddress={false} /></>, detail: withPrice(h.txHash, "On the farm") }];
      case "LotHarvested":
        return h.args.batchId === self
          ? [{ ...base, emoji: "🌾", tone: "field", title: <>Harvested from <Link href={`/batch/${h.args.landId}`} className="underline">land #{h.args.landId}</Link></>, detail: withPrice(h.txHash, `${h.args.quantity} ${unit}`) }]
          : [{ ...base, emoji: "🌾", tone: "field", title: <>Harvested {lotLink(h.args.batchId)}</>, detail: `${h.args.quantity} units` }];
      case "LotSplit":
        return h.args.childId === self
          ? [{ ...base, emoji: "✂️", tone: "sky", title: <>Split from {lotLink(h.args.batchId)}</>, detail: withPrice(h.txHash, `${h.args.quantity} ${unit}`) }]
          : [{ ...base, emoji: "✂️", tone: "sky", title: <>Split off {lotLink(h.args.childId)}</>, detail: `${h.args.quantity} ${unit}` }];
      case "CustodyTransferred":
        return [{ ...base, emoji: "🤝", tone: "sky", title: <>Handed to <PersonName address={h.args.to} showAddress={false} /></>, detail: <>from <PersonName address={h.args.from} showAddress={false} /></> }];
      case "CheckpointAdded":
        return [{ ...base, emoji: "📋", tone: "sky", title: <>Checked by <PersonName address={h.args.inspector} showAddress={false} /></>, detail: checkpointDetail(h.args.dataHash) }];
      case "BatchValidated":
        return [{ ...base, emoji: "✅", tone: "field", title: <>Validated by <PersonName address={h.args.inspector} showAddress={false} /></> }];
      case "PriceRecorded":
        return createdTx.has(h.txHash) ? [] : [{ ...base, emoji: "📈", tone: "field", title: `New last price ${formatFrm(h.args.price)}` }];
    }
  });
}
```

```tsx
import { useState } from "react";
```

```tsx
/** Lot holder: splits the lot into smaller lots that can go to different places. Shows each part's price before signing. */
export function SplitForm({ lot, onDone }: { lot: BatchView; onDone: () => void }) {
  const half = lot.quantity / BigInt(2);
  const [parts, setParts] = useState<string[]>(() => [half.toString(), (lot.quantity - half).toString()]);
  const nums = parts.map((p) => (/^\d+$/.test(p.trim()) ? BigInt(p.trim()) : BigInt(0)));
  const problem = checkSplit(lot.quantity, nums);
  const prices = problem ? [] : splitPrices(lot.lastPrice, nums);

  return (
    <div className="space-y-3">
      <p className="text-sm text-ink-soft">
        This lot has {lot.quantity.toString()} units. Each part becomes its own lot with its share of the price, and this lot closes.
      </p>
      <ul className="space-y-2">
        {parts.map((p, i) => (
          <li key={i} className="flex items-center gap-2">
            <input
              className="input tabular"
              inputMode="numeric"
              aria-label={`Part ${i + 1}`}
              value={p}
              onChange={(e) => setParts(parts.map((x, j) => (j === i ? e.target.value : x)))}
            />
            <span className="w-28 shrink-0 text-right text-sm tabular">{prices[i] !== undefined ? formatFrm(prices[i]) : "…"}</span>
```

```tsx
"use client";

import { useEffect, useState } from "react";
import { formatEther, parseEther } from "ethers";
import { AmountInput, parseFrm } from "@/components/ui/AmountInput";
import { Card } from "@/components/ui/Card";
import { Field } from "@/components/ui/Field";
import { TxButton } from "@/components/ui/TxButton";
import { getContract, getReadProvider } from "@/lib/contracts";
import { formatFrm } from "@/lib/format";
import { mined } from "@/lib/tx";

const RATE = BigInt(1000); // FarmCoin.RATE: 1 ETH = 1000 FRM

type Wallet = { eth: bigint; frm: bigint; supply: bigint; cap: bigint; reserve: bigint };

/** Person B: the acting person's ETH and FRM, buying FRM with ETH and selling it back, and how much of the max supply is used. */
export function WalletCard({ account, version, onDone }: { account: string; version: number; onDone: () => void }) {
  const [w, setW] = useState<Wallet | null>(null);
  const [ethIn, setEthIn] = useState("1");
  const [frmIn, setFrmIn] = useState("");

  useEffect(() => {
    const coin = getContract("FarmCoin");
    if (!coin) return;
    let live = true;
    Promise.all([getReadProvider().getBalance(account), coin.balanceOf(account), coin.totalSupply(), coin.cap(), coin.reserve()])
      .then(([eth, frm, supply, cap, reserve]: bigint[]) => live && setW({ eth, frm, supply, cap, reserve }))
      .catch(() => live && setW(null));
    return () => {
      live = false;
    };
  }, [account, version]);

  let ethWei: bigint | null = null;
  try {
    const v = parseEther(ethIn.trim() || "0");
    ethWei = v > BigInt(0) ? v : null;
  } catch {
    ethWei = null;
  }
  const frmWei = parseFrm(frmIn);
  const used = w && w.cap > BigInt(0) ? Number((w.supply * BigInt(10000)) / w.cap) / 100 : 0;

  return (
    <Card className="space-y-5">
      <div className="flex flex-wrap items-baseline justify-between gap-2">
        <h2 className="font-display text-xl font-bold">Wallet</h2>
        <p className="text-sm text-ink-soft">1 ETH = 1000 FRM, both ways</p>
      </div>
      {w && (
        <p className="text-sm">
          You have <b className="tabular">{formatFrm(w.frm)}</b> and <b className="tabular">{formatEther(w.eth)} ETH</b>
        </p>
      )}
      <div className="grid gap-5 sm:grid-cols-2">
        <section className="space-y-2">
          <Field label="Buy FRM with ETH">
            <input className="input tabular" inputMode="decimal" value={ethIn} onChange={(e) => setEthIn(e.target.value)} />
          </Field>
          <p className="text-xs text-ink-soft">You get {ethWei ? formatFrm(ethWei * RATE) : "…"}</p>
          <TxButton label="Buy FRM" done="FRM bought" variant="primary" disabled={!ethWei} send={(signed) => mined(signed("FarmCoin").buy({ value: ethWei! }))} onDone={onDone} />
        </section>
        <section className="space-y-2">
          <AmountInput label="Sell FRM for ETH" value={frmIn} onChange={setFrmIn} />
          <p className="text-xs text-ink-soft">You get {frmWei ? `${formatEther(frmWei / RATE)} ETH` : "…"}</p>
          <TxButton
            label="Sell FRM"
            done="FRM sold"
            disabled={!frmWei}
            send={(signed) => mined(signed("FarmCoin").sell(frmWei!))}
            onDone={() => {
              setFrmIn("");
              onDone();
            }}
          />
        </section>
      </div>
```
